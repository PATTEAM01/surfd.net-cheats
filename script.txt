    function shade(hex, amt) {
        const n = parseInt(hex.slice(1), 16);
        let r = (n >> 16) & 0xff, g = (n >> 8) & 0xff, b = n & 0xff;
        if (amt > 0) {
            r = r + (255 - r) * amt; g = g + (255 - g) * amt; b = b + (255 - b) * amt;
        } else {
            r = r * (1 + amt); g = g * (1 + amt); b = b * (1 + amt);
        }
        return `rgb(${r|0},${g|0},${b|0})`;

        const seen = new WeakSet();

            seen.add(obj);
            if (obj.isScene) fbTargets.scenes.push(obj);

            if (obj.isCamera) st._camera = obj;
            ['scene','renderer','three','engine','world','view','render','graphics','core','game','camera'].forEach(k => {
                try { if (obj[k] && typeof obj[k] === 'object') scan(obj[k], depth + 1); } catch {}
            });
        }

        ['scene','renderer','three'].forEach(k => {
            try { if (window[k]) scan(window[k], 0); } catch {}

    function applyFullbright() {

                    mats.forEach(m => {

                                m.__dk_origEm = m.emissive.getHex();

        fbTargets.renderers.forEach(r => {

    // ── MAIN STEP HOOK ─────────────────────────────────────────
    const origStep = S.step;
    let spinAccum = 0;

    S.step = function (cmd) {
        cmd = { ...cmd };
        if (!st.startPos && this.state?.position) st.startPos = { ...this.state.position };

        const grounded = this.state?.grounded;
        const dt = this.config?.dt || 0.016;

        if (st.paused) return null;

        if (st.autoRestart && this.state.position.z < -2000 && st.startPos) {
            Object.assign(this.state.position, st.startPos);
            this.state.velocity.x = 0; this.state.velocity.y = 0; this.state.velocity.z = 0;
            toast('Auto-restart', 'success');
        }
        if (st.autoBhop && grounded) cmd.jump = true;
        if (st.autoStrafe && !grounded) {
            const v = this.state.velocity;
            const sp = Math.hypot(v.x, v.y);
            if (sp > 50) {
                const velAng = Math.atan2(v.y, v.x);
                const viewAng = cmd.yaw * Math.PI / 180;
                let delta = viewAng - velAng;
                while (delta > Math.PI) delta -= 2 * Math.PI;
                while (delta < -Math.PI) delta += 2 * Math.PI;
                cmd.side = Math.sign(delta) * 400;
                cmd.forward = 0;
            }
        }
        if (st.spin && !grounded) {
            spinAccum += st.spinSpeed * dt;
            cmd.yaw += spinAccum;
        } else if (!st.spin) spinAccum = 0;

        if (st.flyMode) {
            this.state.velocity.z = 0;
            this.state.grounded = false;
            const spd = 800;
            const yawRad = cmd.yaw * Math.PI / 180;
            const f = (cmd.forward / 400) * spd * dt;
            const s = (cmd.side / 400) * spd * dt;
            this.state.position.x += Math.cos(yawRad) * f + Math.sin(yawRad) * s;
            this.state.position.y += Math.sin(yawRad) * f - Math.cos(yawRad) * s;
            if (cmd.jump) this.state.position.z += spd * dt;
            if (cmd.duck) this.state.position.z -= spd * dt;
        }

        const origDt = this.config.dt;
        this.config.dt = origDt * st.timeScale;
        const result = origStep.call(this, cmd);
        this.config.dt = origDt;

        if (st.speedBoost > 1) {
            const v = this.state.velocity;
            const sp = Math.hypot(v.x, v.y);
            if (sp > 50) { v.x *= st.speedBoost; v.y *= st.speedBoost; }
        }
        if (st.instant && !this.state.grounded) {
            const v = this.state.velocity;
            const sp = Math.hypot(v.x, v.y);
            const targetMax = (cfg.maxSpeed || 300) * 3;
            if (sp > 0.1) { const sc = targetMax / sp; v.x *= sc; v.y *= sc; }
            else {
                const yaw = cmd.yaw * Math.PI / 180;
                v.x = Math.cos(yaw) * targetMax;
                v.y = Math.sin(yaw) * targetMax;
            }
            if (this.config.maxVelocity < targetMax) this.config.maxVelocity = targetMax * 1.5;
        }
        if (st.infJump && cmd.jump && !this.state.grounded) {
            if (this.state.velocity.z < 200) this.state.velocity.z = 250;
        }
        if (st.forceFinish && this.timer.state.phase === 'running') {
            const fz = this.timer.zones.find(z => z.kind === 'finish');
            if (fz) {
                this.state.position.x = (fz.min.x + fz.max.x) / 2;
                this.state.position.y = (fz.min.y + fz.max.y) / 2;
                this.state.position.z = (fz.min.z + fz.max.z) / 2;
                this.state.velocity.x = 0; this.state.velocity.y = 0; this.state.velocity.z = 0;
                origStep.call(this, { ...cmd, forward: 0, side: 0, jump: false, duck: false });
                toast('Force Finish', 'success');
            }
            st.forceFinish = false;
        }
        if (st.minimap) {
            st.trail.push({ x: this.state.position.x, y: this.state.position.y });
            if (st.trail.length > 120) st.trail.shift();
        }
        return result;
    };

    };

    function buildSVG() {
        const shp = SHAPES[kst.type] || SHAPES.karambit;
        const fin = FINISHES[kst.finish] || FINISHES.vanilla;
        const uid = 'dk' + (++svgUid);

        // Blade gradient (diagonal)
        const stops = fin.blade.map((c, i, arr) =>
            `<stop offset="${(i / Math.max(1, arr.length - 1)) * 100}%" stop-color="${c}"/>`
        ).join('');

        // Handle gradient (horizontal — cylindrical)
        const hBase = fin.handle;
        const hStops = `
            <stop offset="0%"   stop-color="${shade(hBase, -0.5)}"/>
            <stop offset="35%"  stop-color="${shade(hBase, 0.15)}"/>
            <stop offset="55%"  stop-color="${shade(hBase, 0.15)}"/>
            <stop offset="100%" stop-color="${shade(hBase, -0.6)}"/>
        `;

        // Guard — slight bevel via 3 vertical stops
        const gStops = `
            <stop offset="0%"   stop-color="${shade(fin.guard, 0.3)}"/>
            <stop offset="50%"  stop-color="${fin.guard}"/>
            <stop offset="100%" stop-color="${shade(fin.guard, -0.4)}"/>
        `;

        // Wear filter — brightness + saturation drop
        const wearStyle = kst.wear > 0.01
            ? `filter: brightness(${1 - kst.wear * 0.35}) saturate(${1 - kst.wear * 0.45}) contrast(${1 - kst.wear * 0.15});`
            : '';

        return `
        <svg viewBox="0 0 300 420" width="100%" height="100%"
             xmlns="http://www.w3.org/2000/svg"
             style="filter: drop-shadow(0 0 14px ${fin.blade[0]}55);">
            <defs>
                <linearGradient id="${uid}-bg" x1="0" y1="0" x2="1" y2="1">
                    ${stops}
                </linearGradient>
                <linearGradient id="${uid}-hg" x1="0" y1="0" x2="1" y2="0">
                    ${hStops}
                </linearGradient>
                <linearGradient id="${uid}-gg" x1="0" y1="0" x2="0" y2="1">
                    ${gStops}
                </linearGradient>
                <clipPath id="${uid}-hc">
                    <path d="${shp.handle}"/>
                </clipPath>
            </defs>

            <!-- BLADE (with wear filter) -->
            <g style="${wearStyle}">
                <path d="${shp.blade}" fill="url(#${uid}-bg)"
                      stroke="rgba(0,0,0,0.55)" stroke-width="1.2" stroke-linejoin="round"/>
                <!-- Left facet darkening (low-poly shading) -->
                <path d="${shp.bladeLeft}" fill="#000" opacity="0.28"/>
                <!-- Tip highlight facet -->
                <path d="${shp.bladeTip}" fill="#fff" opacity="0.08"/>
                <!-- Blood groove -->
                <path d="${shp.groove}" stroke="rgba(0,0,0,0.45)"
                      stroke-width="2.5" stroke-linecap="round" fill="none"/>
                <path d="${shp.groove}" stroke="rgba(255,255,255,0.12)"
                      stroke-width="0.8" stroke-linecap="round" fill="none"
                      transform="translate(-1,-1)"/>
                <!-- Edge highlight -->
                <path d="M150,40 L200,130 L200,250" stroke="rgba(255,255,255,0.42)"
                      stroke-width="1.6" fill="none" stroke-linejoin="round"/>
                <!-- Spine highlight -->
                <path d="M150,40 L105,100" stroke="rgba(255,255,255,0.2)"
                      stroke-width="1.2" fill="none"/>
            </g>

            <!-- HANDLE -->
            <g clip-path="url(#${uid}-hc)">
                <rect x="90" y="260" width="120" height="150" fill="url(#${uid}-hg)"/>
                <!-- Grip rings (5 bands) -->
                <line x1="90" y1="290" x2="210" y2="290" stroke="rgba(0,0,0,0.55)" stroke-width="2.5"/>
                <line x1="90" y1="290" x2="210" y2="290" stroke="rgba(255,255,255,0.08)" stroke-width="0.8" transform="translate(0,-2)"/>
                <line x1="90" y1="316" x2="210" y2="316" stroke="rgba(0,0,0,0.55)" stroke-width="2.5"/>
                <line x1="90" y1="316" x2="210" y2="316" stroke="rgba(255,255,255,0.08)" stroke-width="0.8" transform="translate(0,-2)"/>
                <line x1="90" y1="342" x2="210" y2="342" stroke="rgba(0,0,0,0.55)" stroke-width="2.5"/>
                <line x1="90" y1="342" x2="210" y2="342" stroke="rgba(255,255,255,0.08)" stroke-width="0.8" transform="translate(0,-2)"/>

    function triggerSwing(kind = 'slash') {
        if (!kst.enabled || kst.swinging) return;
        const now = performance.now();
        if (now - kst.lastSwing < 420) return;
        kst.lastSwing = now;
        kst.swinging = true;
        kst.swingKind = kind;
        vm.classList.remove('swinging-slash', 'swinging-stab');
        void vm.offsetWidth;
        vm.classList.add(kind === 'stab' ? 'swinging-stab' : 'swinging-slash');

        const running = S?.timer?.state?.phase === 'running';
        if (kst.statTrakOn && running) {
            kst.statTrak++;
            renderStatTrak();
        }
        setTimeout(() => {
            kst.swinging = false;
            vm.classList.remove('swinging-slash', 'swinging-stab');
        }, kind === 'stab' ? 440 : 400);
    }

    // ── MINIMAP + STATS LOOP ──────────────────────────────────
    function getMinimapBounds() {
        const zones = S.timer.zones || [];
        let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
        zones.forEach(z => {
            minX = Math.min(minX, z.min.x); maxX = Math.max(maxX, z.max.x);
            minY = Math.min(minY, z.min.y); maxY = Math.max(maxY, z.max.y);
        });
        if (!isFinite(minX)) {
            const p = S.state.position;
            minX = p.x - 1000; maxX = p.x + 1000;
            minY = p.y - 1000; maxY = p.y + 1000;
        }
        const padX = (maxX - minX) * 0.15, padY = (maxY - minY) * 0.15;
        return { minX: minX - padX, maxX: maxX + padX, minY: minY - padY, maxY: maxY + padY };
    }

    function drawMinimap() {
        if (!st.minimap) return;
        const W = 180, H = 180;
        const b = getMinimapBounds();
        const w = b.maxX - b.minX, h = b.maxY - b.minY;
        const scale = Math.min(W / w, H / h) * 0.9;
        const offX = (W - w * scale) / 2, offY = (H - h * scale) / 2;
        mmCtx.clearRect(0, 0, W, H);
        mmCtx.fillStyle = 'rgba(6,6,14,0.85)';
        mmCtx.fillRect(0, 0, W, H);
        const toX = x => offX + (x - b.minX) * scale;
        const toY = y => H - offY - (y - b.minY) * scale;

        (S.timer.zones || []).forEach(z => {
            mmCtx.strokeStyle = z.kind === 'finish' ? '#00ff88' :
                                z.kind === 'start' ? '#00e5ff' : '#444';
            mmCtx.lineWidth = 1.5;
            mmCtx.beginPath();
            mmCtx.rect(toX(z.min.x), toY(z.max.y),
                       (z.max.x - z.min.x) * scale, (z.max.y - z.min.y) * scale);
            mmCtx.stroke();
        });
        if (st.trail.length > 1) {
            mmCtx.strokeStyle = 'rgba(185,51,255,0.5)';
            mmCtx.lineWidth = 1.5;
            mmCtx.beginPath();
            st.trail.forEach((p, i) => {
                const x = toX(p.x), y = toY(p.y);
                if (i === 0) mmCtx.moveTo(x, y); else mmCtx.lineTo(x, y);
            });
            mmCtx.stroke();
        }
        const p = S.state.position;
        mmCtx.fillStyle = '#fff';
        mmCtx.beginPath();
        mmCtx.arc(toX(p.x), toY(p.y), 3.5, 0, Math.PI * 2);
        mmCtx.fill();
        mmCtx.strokeStyle = '#b933ff';
        mmCtx.lineWidth = 1.5;
        mmCtx.stroke();
    }
